package q1;

import java.io.File;
import java.io.FileNotFoundException;
import java.io.FileWriter;
import java.io.IOException;
import java.util.ArrayList;
import java.util.Scanner;
import java.util.List;
import java.util.Comparator;

public class Q1 {

    //Change the name of input and output file based on practical paper
    String inputFile = "input.txt";
    String outputFile = "output.txt";

    //--VARIABLES - @STUDENT: DECLARE YOUR VARIABLES HERE:
    StringBuilder output = new StringBuilder();
    ArrayList<String> commands = new ArrayList<>();

    //--FIXED PART - DO NOT EDIT ANY THINGS HERE--
    //--START FIXED PART--------------------------    
    String fi, fo;

    /**
     * Set input and output file for automatic rating
     *
     * @param args path of input file and path of output file
     */
    public void setFile(String[] args) {
        fi = args.length >= 2 ? args[0] : inputFile;
        fo = args.length >= 2 ? args[1] : outputFile;
    }

    /**
     * Reads data from input file
     */
    public void read() {
        try (Scanner sc = new Scanner(new File(fi))) {
            //--END FIXED PART----------------------------

            //INPUT - @STUDENT: ADD YOUR CODE FOR INPUT HERE:
            int n = sc.nextInt();
            sc.nextLine();

            while (sc.hasNextLine()) {
                String line = sc.nextLine().trim();
                if (!line.isEmpty()) {
                    commands.add(line);
                }
            }

            //--FIXED PART - DO NOT EDIT ANY THINGS HERE--
            //--START FIXED PART--------------------------    
            sc.close();
        } catch (FileNotFoundException ex) {
            System.out.println("Input Exception # " + ex);
        }
    }
    //--END FIXED PART----------------------------

    //ALGORITHM - @STUDENT: ADD YOUROWN METHODS HERE (IF NEED):
    //--FIXED PART - DO NOT EDIT ANY THINGS HERE--
    //--START FIXED PART--------------------------    
    /**
     * Main algorithm
     */
    public void solve() {
        //--END FIXED PART----------------------------
        List<Rectangle> listR = new ArrayList<>();
        List<Circle> listC = new ArrayList<>();
        List<Double> listA = new ArrayList<>();

        //ALGORITHM - @STUDENT: ADD YOUR CODE HERE:
        for (String cmd : commands) {
            String[] parts = cmd.split(" ");
            String key = parts[0];

            switch (key) {
                case "C":
                    Circle c = new Circle(Double.parseDouble(parts[3]), parts[1], parts[2]);
                    listC.add(c);
                    listA.add(c.getArea());
                    output.append(c.entryData()).append("\n");
break;

                case "R":
                    Rectangle r = new Rectangle(Double.parseDouble(parts[3]), Double.parseDouble(parts[4]), parts[1], parts[2]);
                    listR.add(r);
                    listA.add(r.getArea());
                    output.append(r.entryData()).append("\n");
                    break;

                case "GetRectangle":
                    output.append("---GetRectangle---\n");
                    if (listR.isEmpty()) {
                        output.append("Empty\n");
                    } else {
                        for (Rectangle r1 : listR) {
                            output.append(r1.printData()).append("\n");
                        }
                    }
                    break;

                case "GetCircle":
                    output.append("---GetCircle---\n");
                    if (listC.isEmpty()) {
                        output.append("Empty\n");
                    } else {
                        for (Circle c1 : listC) {
                            output.append(c1.printData()).append("\n");
                        }
                    }
                    break;

                case "GetMaxArea":
                    output.append("---GetMaxArea---\n");
                    if (listC.isEmpty() && listR.isEmpty()) {
                        output.append("Empty\n");
                        break;
                    }
                    Circle cmax = listC.stream().max((a, b) -> Double.compare(a.getArea(), b.getArea())).orElse(null);
                    Rectangle rmax = listR.stream().max((a, b) -> Double.compare(a.getArea(), b.getArea())).orElse(null);

                    if (cmax.getArea() >= rmax.getArea()) {
                        output.append(cmax.printData()).append("\n");
                    } else {
                        output.append(rmax.printData()).append("\n");
                    }

                    break;

                case "Clear":
                    listR.clear();
                    listC.clear();
                    listA.clear();
                    output.append("------\n* Remove all").append("\n");
                    break;
                case "Sort":
                    output.append("---Sort---").append("\n");
                   
                    if (listC.isEmpty() && listR.isEmpty()) {
                        output.append("Empty\n");
                        break;
                    }                    
                    
                    listA.sort(Double::compareTo);

                    double epsilon = 1e-9;

                    for (double a : listA) {
                        Circle cp = listC.stream()
                                .filter(cs -> Math.abs(cs.getArea() - a) < epsilon)
                                .findFirst()
                                .orElse(null);

                        Rectangle rp = listR.stream().filter(rs -> Math.abs(rs.getArea() - a) < epsilon)
                                .findFirst()
                                .orElse(null);
       
                        if (cp != null) {
                            output.append(cp.printData()).append("\n");
                        }
                        if (rp != null) {
                            output.append(rp.printData()).append("\n");
                        }
                    }
                    break;
                    
                case "Print":
                    output.append("---Print---\n"); 
                   
                    if (listC.isEmpty() && listR.isEmpty()) {
                        output.append("Empty\n");
                        break;
                    }                    

                    double epsilon1 = 1e-9;

                    for (double a : listA) {
                        Circle cp = listC.stream()
                                .filter(cs -> Math.abs(cs.getArea() - a) < epsilon1)
                                .findFirst()
                                .orElse(null);

                        Rectangle rp = listR.stream()
                                .filter(rs -> Math.abs(rs.getArea() - a) < epsilon1)
                                .findFirst()
                                .orElse(null);
                        
                        if (cp != null) {
                            output.append(cp.printData()).append("\n");
                        }
                        if (rp != null) {
                            output.append(rp.printData()).append("\n");
                        }
                    }
                    break;

            }
        }

        //--FIXED PART - DO NOT EDIT ANY THINGS HERE--
        //--START FIXED PART-------------------------- 
    }

    /**
     * Write result into output file
     */
    public void printResult() {
        try {
            FileWriter fw = new FileWriter(fo);
            //--END FIXED PART----------------------------

            //OUTPUT - @STUDENT: ADD YOUR CODE FOR OUTPUT HERE:
            fw.write(output.toString());

            //--FIXED PART - DO NOT EDIT ANY THINGS HERE--
            //--START FIXED PART-------------------------- 
            fw.flush();
            fw.close();
        } catch (IOException ex) {
            System.out.println("Output Exception # " + ex);
        }
    }

    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) {
        // TODO code application logic here
        Q1 q = new Q1();
        q.setFile(args);
        q.read();
        q.solve();
        q.printResult();
    }
    //--END FIXED PART----------------------------    
}
